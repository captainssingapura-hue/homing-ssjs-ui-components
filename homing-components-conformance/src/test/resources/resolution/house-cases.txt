# The house's buttons resolved in the house styles, each written over Base and worn in Harbour: the
# same walk as core's cases, over the house taxonomy.

theme editorial

# the frame is the button's; colour is the palette's, the rank each button's
plain-button         rest      shape-corner       => button__shape-corner from editorial
danger-button        rest      shape-corner       => button__shape-corner from editorial
primary-button       rest      color-background   => primary-button__color-background from harbour
secondary-button     rest      color-ink          => secondary-button__color-ink from harbour
warning-button       rest      color-border       => button__color-border from harbour
# a box's shadow is drawn in the palette's colours, set on the element itself
danger-button        rest      color-box-shadow   => root__color-box-shadow from harbour
# what every style says alike is Base's
plain-button         rest      type-scale         => button__type-scale from base
plain-button         rest      affordance-cursor  => control__affordance-cursor from base
# Go on the primary button, as the action bar casts it: nothing says more of it than of the button
primary-button_go    rest      type-weight        => root__type-weight from base
# the plain button in the picker's roles: nothing is said of the roles, or of the plain button - a button's face
plain-button_choice  rest      color-background   => button__color-background from harbour
plain-button_cancel  rest      shape-corner       => button__shape-corner from editorial
# treatments, common to every role: the ring, the lift, how it is when it cannot be used
plain-button_cancel  disabled  effect-opacity     => control__effect-opacity--disabled from editorial
danger-button        hover     shape-shadow       => button__shape-shadow--hover from editorial
danger-button        disabled  shape-shadow       => control__shape-shadow--disabled from base
danger-button        focus     shape-outline      => control__shape-outline--focus from editorial
danger-button        focus     color-outline      => control__color-outline--focus from harbour
danger-button        focus     shape-shadow       => none
# a functional role is set aside: the dialog's window looks as the floating pane on its own
floating-pane_window rest      shape-corner       => card__shape-corner from editorial
# a toggle that is on is held pressed, tinted with the accent; the close button draws the style's own cross
toggle-button        pressed   shape-shadow       => toggle-button__shape-shadow--pressed from editorial
toggle-button        pressed   color-background   => toggle-button__color-background--pressed from harbour
toggle-button        hover     shape-shadow       => button__shape-shadow--hover from editorial
close-button         rest      type-glyph         => root__type-glyph from base

theme sketchy

# ink casts no shadow: sketchy says nothing of it, and Base says none
danger-button        rest      shape-shadow       => root__shape-shadow from base
danger-button        hover     shape-shadow       => none
danger-button        disabled  shape-shadow       => control__shape-shadow--disabled from base
danger-button        rest      shape-corner       => button__shape-corner from sketchy
toggle-button        pressed   shape-shadow       => none

theme neumorphism

# moulded, not ruled: its ring is its own, its border Base's
plain-button         rest      shape-border       => root__shape-border from base
plain-button         focus     shape-outline      => control__shape-outline--focus from neumorphism
plain-button         active    motion-transform   => none

theme neo-brutalism

# pressed into its own shadow
plain-button         active    motion-transform   => button__motion-transform--active from neo-brutalism

theme editorial_forest

# the same style in another palette: its shadow and frame its own, every colour the forest's
danger-button        hover     shape-shadow       => button__shape-shadow--hover from editorial
danger-button        rest      color-box-shadow   => root__color-box-shadow from forest
primary-button       rest      color-background   => primary-button__color-background from forest
